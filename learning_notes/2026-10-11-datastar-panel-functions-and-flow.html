Background: two systems

If you know Django and the DOM, continue to the next section. This section explains the two systems that meet inside the panel.

Django makes an HTTP response. Django Debug Toolbar adds a panel to that response. Datastar runs later in the browser and owns the signal store.

The server cannot inspect later browser signals through the first HTTP response. This package sends the panel shell to the browser, then reads the signals there.

→
Debug Toolbar

→
Browser module

src/django_datastar_toolbar/panels.py registers the panel. The HTML template makes the status, search, and output elements. The CSS positions the panel and marks search results.

The wheel ships the three JavaScript modules. Django Debug Toolbar loads datastar_store.js. That file imports display.js and search.js.

Core idea: one signal change

Imagine a Datastar store with count: 2. The panel script adds a hidden pre element to the page body. Its data-json-signals__terse attribute tells Datastar to write compact JSON into the element's text.

The hidden collector lives in the page body. The visible output lives inside the toolbar. This matters when the toolbar uses a shadow root.

The panel shows a current JSON projection. It does not keep a list of earlier signal changes.

currentToolbar()

Find the toolbar in a normal DOM host or its shadow root. Return null if the toolbar does not exist.

function currentToolbar() {
  // Find the current Debug Toolbar in the host or its shadow root.
  const host = document.getElementById("djDebugRoot");
  return (host?.shadowRoot ?? host)?.querySelector("#djDebug") ?? null;
}

content()

Find this panel inside the current toolbar. Other functions use this value instead of an old panel node.

function content() {
  // Find this panel's content in the current toolbar.
  return (
    toolbar?.querySelector(`#${panelId} [data-datastar-store-panel]`) ??
    null
  );
}

watchBody()

Watch for a body change, then ask sync() to find the toolbar. A new body needs a new observer.

function watchBody() {
  // Watch body changes to find new toolbar content.
  bodyObserver?.disconnect();
  bodyObserver = new MutationObserver(sync);
  if (document.body) bodyObserver.observe(document.body, { childList: true });
  sync();
}

sync()

Keep event listeners and one collector on the active panel. Stop old state when the toolbar or its panel changes.

function sync() {
  // Attach to the current toolbar and keep one collector for its panel.
  const nextToolbar = currentToolbar();
  if (nextToolbar !== toolbar) {
    stopCollector();
    shellObserver?.disconnect();
    toolbar?.removeEventListener("djdt.panel.render", onPanelRender);
    toolbar?.removeEventListener("input", onInput);
    toolbar?.removeEventListener("keydown", onKeyDown);
    toolbar = nextToolbar;
    if (toolbar) {
      toolbar.addEventListener("djdt.panel.render", onPanelRender);
      toolbar.addEventListener("input", onInput);
      toolbar.addEventListener("keydown", onKeyDown);
      shellObserver = new MutationObserver(sync);
      shellObserver.observe(toolbar, { childList: true, subtree: true });
    }
  }
  const panel = content();
  if (!panel) {
    stopCollector();
    return;
  }
  if (panel !== attachedContent) {
    stopCollector();
    attachedContent = panel;
    startCollector();
  } else {
    render();
  }
}

onPanelRender()

React only when the toolbar adds this panel. A different panel does not need a new collector.

function onPanelRender(event) {
  // Check this panel when the toolbar adds its HTML.
  if (event.detail?.panelId === panelId) sync();
}

startCollector()

Add a hidden element to the light DOM, then watch its text. Show waiting text until Datastar writes JSON.

function startCollector() {
  // Create the hidden JSON collector and watch it for changes.
  setStatus(content(), "Waiting for Datastar signal data…");
  collector = document.createElement("pre");
  collector.hidden = true;
  collector.setAttribute("data-json-signals__terse", "");
  collector.setAttribute("data-djdt-datastar-collector", "");
  // Datastar observes the host's light DOM, not the toolbar's shadow root.
  document.body.append(collector);
  collectorObserver = new MutationObserver(render);
  collectorObserver.observe(collector, {
    subtree: true,
    childList: true,
    characterData: true,
  });
  pendingTimer = setTimeout(() => {
    if (collector && collector.textContent === "") {
      setStatus(
        content(),
        "Still waiting for Datastar signal data…"
      );
    }
  }, waitMs);
  render();
}

stopCollector()

Stop the timer, disconnect the observer, remove the collector, and clear search state. This lets a replacement panel start clean.

function stopCollector() {
  // Remove the collector and its observer, then clear panel state.
  clearTimeout(pendingTimer);
  collectorObserver?.disconnect();
  collectorObserver = null;
  collector?.remove();
  collector = null;
  attachedContent = null;
  projectionText = "";
  resetSearch();
}

Module 2: build the display

The collector text changes when Datastar writes a new JSON projection. render() connects that text to the visible output.

render()

Ignore missing or unchanged text. Parse new JSON, show an error for invalid JSON, and refresh search after a valid display.

function render() {
  // Read new collector text, parse it, and show the result or an error.
  const panel = content();
  if (!panel || !collector) return;
  const text = collector.textContent;
  if (!text || text === projectionText) return;
  let projection;
  try {
    projection = JSON.parse(text);
  } catch {
    projectionText = "";
    clearTimeout(pendingTimer);
    setStatus(
      panel,
      "Invalid JSON projection. The panel cleared the previous display."
    );
    setText(panel.querySelector("[data-datastar-store-output]"), "");
    updateMatches(panel, Boolean(projectionText));
    return;
  }
  projectionText = text;
  clearTimeout(pendingTimer);
  setStatus(panel, "");
  if (renderProjection(panel, projection))
    updateMatches(panel, Boolean(projectionText));
}

display.js owns DOM output. It uses textContent and DOM nodes, not HTML strings from signal values.

setText()

Change text only when its value differs. This helps the status and search count avoid extra DOM changes.

export function setText(node, text) {
  // Write text only when it changes, so the DOM gets fewer changes.
  if (node && node.textContent !== text) node.textContent = text;
}

setStatus()

Put a message in the panel status line. Hide the line when the message is empty.

export function setStatus(panel, text) {
  // Set the status text and hide the line when it is empty.
  const status = panel?.querySelector("[data-datastar-store-status]");
  setText(status, text);
  if (status) status.hidden = !text;
}

isBranch()

Tell the renderer whether a parsed JSON value has children. Arrays and objects have children, but null does not.

function isBranch(value) {
  // Treat arrays and non-null objects as JSON branches.
  return value !== null && typeof value === "object";
}

entries()

Give arrays numeric indexes and objects their own keys. The renderer can then use one loop for both types.

function entries(value) {
  // Pair array items with indexes, or get object key-value pairs.
  return Array.isArray(value)
    ? value.map((item, index) => [index, item])
    : Object.entries(value);
}

displayPath()

Build a path such as items[0].name. Quote unusual object keys so the path stays readable.

function displayPath(parent, key) {
  // Build a readable path for an object key or array index.
  if (typeof key === "number") return `${parent}[${key}]`;
  const part = /^[a-zA-Z_$][\w$]*$/.test(key)
    ? key
    : `[${JSON.stringify(key)}]`;
  return parent
    ? part.startsWith("[")
      ? `${parent}${part}`
      : `${parent}.${part}`
    : part;
}

keyMarker()

Make a span with a full path and key name in data attributes. Search uses these markers without hiding values.

function keyMarker(key, path) {
  // Mark a key with its path so search can find it.
  const marker = document.createElement("span");
  marker.dataset.datastarStorePath = path;
  marker.dataset.datastarStoreKey =
    typeof key === "number" ? `[${key}]` : key;
  return marker;
}

appendJsonValue()

Walk each JSON branch and write indented text. Add markers beside keys and array items for search.

function appendJsonValue(parent, value, path, depth, marker) {
  // Walk the JSON value and add text and path markers to the display.
  if (!isBranch(value)) {
    const text = JSON.stringify(value);
    if (marker) {
      marker.textContent = text;
      parent.append(marker);
    } else {
      parent.append(text);
    }
    return;
  }

  const array = Array.isArray(value);
  const children = entries(value);
  const opening = array ? "[" : "{";
  if (marker) {
    marker.textContent = opening;
    parent.append(marker);
  } else {
    parent.append(opening);
  }
  for (const [index, [key, child]] of children.entries()) {
    const childPath = displayPath(path, key);
    const childMarker = keyMarker(key, childPath);
    parent.append("\n", "  ".repeat(depth + 1));
    if (array) {
      appendJsonValue(parent, child, childPath, depth + 1, childMarker);
    } else {
      childMarker.textContent = JSON.stringify(key);
      parent.append(childMarker, ": ");
      appendJsonValue(parent, child, childPath, depth + 1);
    }
    if (index < children.length - 1) parent.append(",");
  }
  if (children.length) parent.append("\n", "  ".repeat(depth));
  parent.append(array ? "]" : "}");
}

renderProjection()

Build new DOM in a fragment and replace the old output. Return false when the panel lacks an output element.

export function renderProjection(panel, value) {
  // Build the new display before replacing the old display.
  const output = panel.querySelector("[data-datastar-store-output]");
  if (!output) return false;
  const fragment = document.createDocumentFragment();
  appendJsonValue(fragment, value, "", 0);
  output.replaceChildren(fragment);
  output.hidden = false;
  return true;
}

onInput()

Pass toolbar input events to search with the active panel and projection status.

function onInput(event) {
  // Give search the active panel and its projection status.
  handleInput(event, content(), Boolean(projectionText));
}

handleInput()

Change matches only when the event comes from this panel’s search input.

export function handleInput(event, panel, hasProjection) {
  // Refresh the search results only when the search field changes.
  if (event.target === panel?.querySelector("[data-datastar-store-search]"))
    updateMatches(panel, hasProjection);
}

updateMatches()

Reset the selected match, check key names and paths, and mark matching spans. Keep every value visible.

export function updateMatches(panel, hasProjection) {
  // Mark matching names and paths without hiding other values.
  const count = panel.querySelector("[data-datastar-store-match-count]");
  const query = panel
    .querySelector("[data-datastar-store-search]")
    ?.value.trim()
    .toLocaleLowerCase();
  matches = [];
  currentMatch = 0;
  if (count) count.hidden = !query || !hasProjection;
  for (const marker of panel.querySelectorAll("[data-datastar-store-path]")) {
    const matchesQuery =
      Boolean(query) &&
      (marker.dataset.datastarStorePath
        .toLocaleLowerCase()
        .includes(query) ||
        marker.dataset.datastarStoreKey
          .toLocaleLowerCase()
          .includes(query));
    marker.classList.toggle("djdt-datastar-match", matchesQuery);
    marker.classList.remove("djdt-datastar-current-match");
    if (matchesQuery) matches.push(marker);
  }
  if (query && hasProjection) showCurrentMatch(panel);
}

showCurrentMatch()

Show the match number, mark one result as current, and scroll to it when the panel is visible.

function showCurrentMatch(panel) {
  // Mark the selected match, show its count, and scroll to it.
  for (const match of matches)
    match.classList.remove("djdt-datastar-current-match");
  const count = panel.querySelector("[data-datastar-store-match-count]");
  setText(
    count,
    matches.length
      ? `${currentMatch + 1} of ${matches.length} matching paths (Enter for next)`
      : "No matching signal names or paths."
  );
  if (matches.length) {
    matches[currentMatch].classList.add("djdt-datastar-current-match");
    if (panel.getClientRects().length)
      matches[currentMatch].scrollIntoView({ block: "nearest" });
  }
}

onKeyDown()

Pass toolbar key events to the search module with the active panel.

function onKeyDown(event) {
  // Give search the active panel for keyboard navigation.
  handleKeyDown(event, content());
}

handleKeyDown()

Use Enter to select the next match. Use Shift+Enter to select the previous match.

export function handleKeyDown(event, panel) {
  // Use Enter to move through matches, and Shift+Enter to move back.
  if (
    event.key !== "Enter" ||
    event.target !== panel?.querySelector("[data-datastar-store-search]") ||
    !matches.length
  )
    return;
  event.preventDefault();
  currentMatch =
    (currentMatch + (event.shiftKey ? -1 : 1) + matches.length) %
    matches.length;
  showCurrentMatch(panel);
}

resetSearch()

Clear old match references when the panel goes away. The next panel can make its own results.

export function resetSearch() {
  // Clear matches when the toolbar changes or removes the panel.
  matches = [];
  currentMatch = 0;
}
